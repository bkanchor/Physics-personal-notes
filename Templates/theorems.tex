% Theorem System
% The following boxes are provided:
%   Definition:     \defn 
%   Theorem:        \thm 
%   Principle:      \pr
%   Result:         \res
%   Corollary:      \cor
%   Proposition:    \prop   
%   Claim:          \clm
%   Fact:           \fact
%   Proof:          \pf
%   Example:        \ex
%   Note:           \nt
%   Notation:       \ntn
%   Terminalogy:    \tmg
%   Remark:         \rmk (sentence), \rmkb (block)
% Suffix
%   r:              Allow Theorem/Definition to be referenced, e.g. thmr
%   p:              Add a short proof block for Lemma, Corollary, Proposition or Claim, e.g. lemp
%                   For theorems, use \pf for proof blocks

% Definition
\newtcbtheorem[number within=section]{mydefinition}{Definition}
{
    enhanced,
    frame hidden,
    titlerule=0mm,
    toptitle=1mm,
    bottomtitle=1mm,
    fonttitle=\bfseries\large,
    coltitle=black,
    colbacktitle=red!20!white,
    colback=red!10!white,
}{defn}

\NewDocumentCommand{\defn}{m+m}{
    \begin{mydefinition}{#1}{}
        #2
    \end{mydefinition}
}

\NewDocumentCommand{\defnr}{mm+m}{
    \begin{mydefinition}{#1}{#2}
        #3
    \end{mydefinition}
}

% Example with counter

% --- in the preamble (before \begin{document}) ---
\newcounter{example}[chapter]
\renewcommand{\theexample}{\thechapter.\arabic{example}}

\newenvironment{example}{
    \refstepcounter{example}%
    {\noindent{\it \textbf{Example \theexample}}}
    \tcolorbox[blanker, breakable, left=5mm, right=1mm, parbox=false,
        before upper={\parindent15pt},
        after skip=10pt,
        borderline west={1mm}{0pt}{red!10!white}]
}{
    \textcolor{red!10!white}{\hbox{}\nobreak\hfill$\blacksquare$} 
    \endtcolorbox
}

\NewDocumentCommand{\ex}{+m}{
    \begin{example}
        #1
    \end{example}
}
